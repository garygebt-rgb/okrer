```typescript
export type JobCategory = '研发' | '产品' | '测试' | '管理';

export interface OKRCycle {
  id: string;
  name: string;
  startTime: string;
  endTime: string;
  objectives: Objective[];
}

export interface Objective {
  id: string;
  title: string;
  description: string;
  weight: number;
  score: number;
  keyResults: KeyResult[];
  alignments: Alignment[];
}

export interface KeyResult {
  id: string;
  title: string;
  description: string;
  weight: number;
  score: number;  // 0-1 range from Feishu OKR system
  metrics: string[];
}

export interface Alignment {
  type: 'contributing' | 'dependent';  // 贡献型 vs 依赖型
  parentObjectiveId?: string;
  peerObjectiveId?: string;
}

export interface SmartDimension {
  dimension: 'Specific' | 'Measurable' | 'Achievable' | 'Relevant' | 'Time-bound';
  result: 'pass' | 'issue' | 'suggest';
  currentState: string;
  problem?: string;
  rewriteExample?: string;
}

export interface ScoringBreakdown {
  difficulty: number;    // 0-100
  effort: number;        // 0-100
  completion: number;    // 0-100
  total: number;         // weighted sum
}

export interface EvidenceMatchResult {
  url: string;
  accessible: boolean;
  creatorMatches: boolean;
  withinTimeRange: boolean;
  keywordRelevance: number;  // 0-1
  consistency: boolean;
  overallScore: number;      // 0-1
}
```
